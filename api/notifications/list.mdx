---
title: "List notifications"
description: "Get the authenticated user's in-app notifications."
api: "GET /notifications"
---

Returns a paginated list of the current user's in-app notifications, newest first.

Page through the list with `limit` and `offset` until `has_more` is `false`.

## Query parameters

<ParamField query="limit" type="integer" default="50">
  Number of notifications to return. Min `1`, max `100`.
</ParamField>

<ParamField query="offset" type="integer" default="0">
  Number of notifications to skip.
</ParamField>

<ParamField query="unread_only" type="boolean" default="false">
  Return only notifications that have not been read yet.
</ParamField>

## Response

<ResponseField name="notifications" type="array">
  <Expandable title="NotificationResponse">
    <ResponseField name="id" type="string">Notification UUID.</ResponseField>
    <ResponseField name="user_id" type="string">Recipient user UUID.</ResponseField>
    <ResponseField name="channel" type="string">Delivery channel: `in_app`, `email`.</ResponseField>
    <ResponseField name="subject" type="string">Notification title.</ResponseField>
    <ResponseField name="body" type="string">Notification content.</ResponseField>
    <ResponseField name="status" type="string">Delivery status: `pending`, `sent`, `delivered`, `failed`.</ResponseField>
    <ResponseField name="priority" type="string">Priority: `low`, `normal`, `high`.</ResponseField>
    <ResponseField name="created_at" type="string">ISO 8601 creation timestamp.</ResponseField>
    <ResponseField name="sent_at" type="string">When the notification was sent. `null` if not yet sent.</ResponseField>
    <ResponseField name="delivered_at" type="string">When delivery was confirmed. `null` if not confirmed.</ResponseField>
    <ResponseField name="read_at" type="string">When the user read the notification. `null` if unread.</ResponseField>
    <ResponseField name="metadata" type="object">
      What the notification is about. Every field is `null` (or empty) when it does not apply.
      <Expandable title="metadata">
        <ResponseField name="event_type" type="string">What raised it, e.g. `order_update`, `order_provisioned`, `wallet_update`, `ip_check`.</ResponseField>
        <ResponseField name="order_id" type="string">The order it is about.</ResponseField>
        <ResponseField name="kind" type="string">Which order event it is, e.g. `order_paid`, `order_expired`.</ResponseField>
        <ResponseField name="check_id" type="string">The IP check it is about. Fetch the report with [Get a check](/api/ip-check/get-check).</ResponseField>
        <ResponseField name="message_key" type="string">
          Which message it is, e.g. `order_paid`, `balance_deposited`, `ip_check_completed`. Use it with
          `message_params` to show the notification in your own language and wording; `subject` and
          `body` are the same message in English. `null` on notifications created before keys existed.
        </ResponseField>
        <ResponseField name="message_params" type="object">
          The values the message is filled in with, e.g. `{"order_id": "…"}` or
          `{"ip": "203.0.113.7", "verdict": "clean", "check_id": "…"}`. Strings and integers only.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="total" type="integer">
  How many notifications the query matches across all pages: all of them, or only the unread ones
  with `unread_only=true`.
</ResponseField>

<ResponseField name="unread_count" type="integer">
  Number of unread notifications.
</ResponseField>

<ResponseField name="has_more" type="boolean">
  Whether another page follows this one.
</ResponseField>
